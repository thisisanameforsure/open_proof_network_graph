Notation: S = sum over n of omega(n)/2^n, and for N a natural number
T(N) = sum over j >= 0 of omega(N+1+j)/2^(j+1), the "rescaled tail" in the hole's statement.

1. `omega_series_summable`: the series S is summable (copied from the merged partial on
   `erdos-69--h2-v2`).
2. `tail_zero`: T(0) = S.
3. `hole_of_irrational`: if S is irrational then the hole's conclusion holds for every b, with N = 0.
   The merged assembly on `erdos-69--h2-v2` goes the other way (the hole gives irrationality), so
   the hole is equivalent to the root and the route "N = 0" is circular.
4. `tail_recurrence`: T(N+1) = 2*T(N) - omega(N+1). Its proof contains the summability of the
   shifted, rescaled tail, which any proof of the hole will need.
5. `integrality_step`: if b*T(N) is an integer then so is b*T(N+1). So the set of N at which the
   hole's inequality fails for a given b is closed upwards, and the hole holds for b if and only if
   it holds for b with N = 0. Choosing N is therefore not a matter of series algebra; it has to
   come from arithmetic information about omega on consecutive integers, which I did not attempt.
6. `root_series_eq`: the root's series (from n = 2) equals S.

import Mathlib

open scoped ArithmeticFunction.omega

/-- Summability of the series (copied from the merged partial on `erdos-69--h2-v2`). -/
theorem omega_series_summable : Summable (fun n : ℕ => (ω n : ℝ) / 2 ^ n) := by
  have hbound : ∀ n : ℕ,
      (ω n : ℝ) / 2 ^ n ≤ (n : ℝ) ^ 1 * (1 / 2 : ℝ) ^ n + (1 / 2 : ℝ) ^ n := by
    intro n
    have h1 : ω n ≤ n + 1 := by
      rw [ArithmeticFunction.cardDistinctFactors_apply]
      calc n.primeFactorsList.dedup.length = n.primeFactors.card := rfl
        _ ≤ (Finset.range (n + 1)).card :=
            Finset.card_le_card (fun p hp =>
              Finset.mem_range.2 (Nat.lt_succ_of_le (Nat.le_of_mem_primeFactors hp)))
        _ = n + 1 := Finset.card_range _
    have h2 : (ω n : ℝ) ≤ (n : ℝ) + 1 := by exact_mod_cast h1
    have h4 : (0 : ℝ) ≤ ((2 : ℝ) ^ n)⁻¹ := by positivity
    rw [pow_one, one_div, inv_pow, div_eq_mul_inv]
    nlinarith [mul_le_mul_of_nonneg_right h2 h4]
  refine Summable.of_nonneg_of_le (fun n => by positivity) hbound ?_
  exact (summable_pow_mul_geometric_of_norm_lt_one 1 (by norm_num [Real.norm_eq_abs])).add
    (summable_geometric_of_lt_one (by norm_num) (by norm_num))

/-- Fact 1: the rescaled tail at `N = 0` is the whole series. -/
theorem tail_zero :
    ∑' j : ℕ, (ω (0 + 1 + j) : ℝ) / 2 ^ (j + 1) = ∑' n : ℕ, (ω n : ℝ) / 2 ^ n := by
  have h1 : ∑' n : ℕ, (ω n : ℝ) / 2 ^ n = ∑' n : ℕ, (ω (n + 1) : ℝ) / 2 ^ (n + 1) := by
    rw [omega_series_summable.tsum_eq_zero_add]
    simp
  rw [h1]
  refine tsum_congr (fun j => ?_)
  rw [show 0 + 1 + j = j + 1 by omega]

/-- Fact 2: the hole's conclusion follows from irrationality of the series, with `N = 0`.
Together with the merged assembly of `erdos-69--h2-v2` (which derives irrationality from the
hole), the hole is *equivalent* to the root's irrationality claim. -/
theorem hole_of_irrational
    (hirr : Irrational (∑' n : ℕ, (ω n : ℝ) / 2 ^ n)) :
    ∀ b : ℕ, 0 < b → ∃ N : ℕ, ∀ z : ℤ,
      (b : ℝ) * ∑' j : ℕ, (ω (N + 1 + j) : ℝ) / 2 ^ (j + 1) ≠ (z : ℝ) := by
  intro b hb
  refine ⟨0, fun z hz => ?_⟩
  rw [tail_zero] at hz
  exact (hirr.natCast_mul hb.ne').ne_int z hz

/-- Fact 3: the rescaled tails obey `T (N+1) = 2 * T N - ω (N+1)`. -/
theorem tail_recurrence (N : ℕ) :
    ∑' j : ℕ, (ω (N + 1 + 1 + j) : ℝ) / 2 ^ (j + 1)
      = 2 * ∑' j : ℕ, (ω (N + 1 + j) : ℝ) / 2 ^ (j + 1) - (ω (N + 1) : ℝ) := by
  have hshift : Summable (fun j : ℕ => (ω (j + (N + 1)) : ℝ) / 2 ^ (j + (N + 1))) :=
    (summable_nat_add_iff (f := fun n : ℕ => (ω n : ℝ) / 2 ^ n) (N + 1)).2 omega_series_summable
  have hT : Summable (fun j : ℕ => (ω (N + 1 + j) : ℝ) / 2 ^ (j + 1)) := by
    refine (hshift.mul_left ((2 : ℝ) ^ N)).congr (fun j => ?_)
    rw [show j + (N + 1) = N + 1 + j by omega]
    rw [show (2 : ℝ) ^ (N + 1 + j) = 2 ^ N * 2 ^ (j + 1) by rw [← pow_add]; congr 1; omega]
    field_simp
  rw [hT.tsum_eq_zero_add, mul_add, ← tsum_mul_left]
  have hterm : ∀ j : ℕ, 2 * ((ω (N + 1 + (j + 1)) : ℝ) / 2 ^ (j + 1 + 1))
      = (ω (N + 1 + 1 + j) : ℝ) / 2 ^ (j + 1) := by
    intro j
    rw [show N + 1 + (j + 1) = N + 1 + 1 + j by omega, pow_succ (2 : ℝ) (j + 1)]
    field_simp
  simp_rw [hterm]
  simp only [add_zero, zero_add, pow_one]
  ring

/-- Fact 4: integrality of `b * T N` propagates from `N` to `N + 1`, so the choice of `N`
cannot matter by itself: `b * T N` is an integer for some `N` iff it is for all larger `N`. -/
theorem integrality_step (b N : ℕ)
    (h : ∃ z : ℤ, (b : ℝ) * ∑' j : ℕ, (ω (N + 1 + j) : ℝ) / 2 ^ (j + 1) = (z : ℝ)) :
    ∃ z : ℤ, (b : ℝ) * ∑' j : ℕ, (ω (N + 1 + 1 + j) : ℝ) / 2 ^ (j + 1) = (z : ℝ) := by
  obtain ⟨z, hz⟩ := h
  refine ⟨2 * z - (b : ℤ) * (ω (N + 1) : ℤ), ?_⟩
  rw [tail_recurrence N]
  push_cast
  linear_combination 2 * hz

/-- Fact 5: the root's series (from n = 2) is the same real number. -/
theorem root_series_eq :
    ∑' n : ℕ, (ω (n + 2) : ℝ) / 2 ^ (n + 2) = ∑' n : ℕ, (ω n : ℝ) / 2 ^ n := by
  have h := omega_series_summable.sum_add_tsum_nat_add 2
  rw [← h]
  simp [Finset.sum_range_succ]
